cmake_minimum_required(VERSION 3.21)
project(crt-clock VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network DBus)
add_executable(crt-clock src/main.cpp)
target_link_libraries(crt-clock PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus)
target_compile_options(crt-clock PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS crt-clock RUNTIME DESTINATION bin)
install(FILES packaging/crt-clock.desktop DESTINATION share/applications)
